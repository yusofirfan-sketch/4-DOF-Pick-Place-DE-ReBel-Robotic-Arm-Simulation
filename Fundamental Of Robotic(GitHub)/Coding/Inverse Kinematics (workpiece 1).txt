% Project Robotics Fanuc 4-DOF: Integrated Trajectory & Simulation
clear; clc; close all;
% 1. Physical Parameters (mm) 
d1 = 742; a1 = 260; a2 = 945; a3 = 1025; a4 = 220; d4 = 252;
% 2. Target Coordinates [Px, Py, Pz]
Rest_coord = [a1+a2+a3, 0, d1+d4];
WP1_coord  = [-200, 20, 370.5];    % Koordinat WP1
Rack2_coord = [-180, 70, 550];     % Koordinat Rack 2
targets = {Rest_coord, WP1_coord, Rack2_coord};
q_pts = zeros(3, 4);
% 3. Inverse Kinematics (IK) - Solve for Theta 1 to 4
for i = 1:3
   Px = targets{i}(1); Py = targets{i}(2); Pz = targets{i}(3);
   th1 = atan2(Py, Px);
   dist_xy = sqrt(Px^2 + Py^2);
   R = dist_xy - a1;
   Z_eff = Pz - d1 - d4;
   dist_sq = R^2 + Z_eff^2;
   cos_th3 = max(min((dist_sq - a2^2 - a3^2) / (2 * a2 * a3), 1), -1);
   th3 = acos(cos_th3);
   th2 = atan2(Z_eff, R) + atan2(a3*sin(th3), a2 + a3*cos(th3));
   th4 = -(th2 + th3);
   q_pts(i, :) = [rad2deg(th1), rad2deg(th2), rad2deg(th3), rad2deg(th4)];
end
q_pts(1, :) = [0, 0, 0, 0]; % Reset Home position
% 4. Data Preparation for Table & Graphs
T = 5; dt = 0.1;
t_total = 0:dt:10;
Q_sim = [];
% Logic Generate Trajectory
for j = 1:4
   % Fasa 1 (0s - 5s)
   a2_1 = (3/T^2)*(q_pts(2,j)-q_pts(1,j));
   a3_1 = (-2/T^3)*(q_pts(2,j)-q_pts(1,j));
   q_f1 = q_pts(1,j) + a2_1*(0:dt:T).^2 + a3_1*(0:dt:T).^3;
  
   % Fasa 2 (5s - 10s)
   a2_2 = (3/T^2)*(q_pts(3,j)-q_pts(2,j));
   a3_2 = (-2/T^3)*(q_pts(3,j)-q_pts(2,j));
   q_f2 = q_pts(2,j) + a2_2*(dt:dt:T).^2 + a3_2*(dt:dt:T).^3;
  
   Q_sim(:,j) = [q_f1, q_f2]';
end
% 5. Print Table to Command Window
fprintf('====================================================================================\n');
fprintf(' Time (t) | th1 (deg) | th2 (deg) | th3 (deg) | th4 (deg) |         Status          \n');
fprintf('------------------------------------------------------------------------------------\n');
for t_idx = 0:10
   idx = find(abs(t_total - t_idx) < 1e-5, 1);
   q_row = Q_sim(idx, :);
   status = 'moving';
   if t_idx == 0, status = 'REST'; end
   if t_idx == 5, status = 'AT WP1'; end
   if t_idx == 10, status = 'AT RACK 2'; end
   fprintf('   %2ds      | %8.2f  | %8.2f  | %8.2f  | %8.2f  |  %s \n', t_idx, q_row(1), q_row(2), q_row(3), q_row(4), status);
end
fprintf('====================================================================================\n');
% 6. Plotting Graphs
figure('Name', 'Joint Trajectories (WP1 to Rack 2)');
for j = 1:4
   subplot(2,2,j);
   plot(t_total, Q_sim(:,j), 'LineWidth', 2, 'Color', [0 .45 .74]);
   grid on; title(['Joint ', num2str(j)]); xlabel('Time (s)'); ylabel('Angle (deg)');
end
% 7. 3D Simulation
figure('Name', 'Fanuc 4-DOF Simulation: WP1 to Rack 2');
hold on; grid on; axis equal; view(3);
xlim([-1500 1500]); ylim([-1000 1000]); zlim([0 2000]);
h_arm = plot3(0,0,0, '-o', 'LineWidth', 5, 'Color', 'b', 'MarkerFaceColor', 'k');
h_info = text(-1300, 0, 1800, '', 'FontSize', 10, 'FontWeight', 'bold');
plot3(WP1_coord(1), WP1_coord(2), WP1_coord(3), 'go', 'MarkerSize', 10, 'LineWidth', 2);
plot3(Rack2_coord(1), Rack2_coord(2), Rack2_coord(3), 'mo', 'MarkerSize', 10, 'LineWidth', 2);
tic;
for k = 1:length(t_total)
   q = deg2rad(Q_sim(k,:));
   % Forward Kinematics for Visual
   p0 = [0, 0, 0]; p1 = [0, 0, d1];
   p2 = p1 + [a1*cos(q(1)), a1*sin(q(1)), 0];
   p3 = p2 + [a2*cos(q(1))*cos(q(2)), a2*sin(q(1))*cos(q(2)), a2*sin(q(2))];
   p4 = p3 + [a3*cos(q(1))*cos(q(2)+q(3)), a3*sin(q(1))*cos(q(2)+q(3)), a3*sin(q(2)+q(3))];
  
   % End Effector Position (Sesuai dengan matriks report kau)
   p5_x = cos(q(1))*(a3*cos(q(2)+q(3)) + a2*cos(q(2)) + a1) + d4*sin(q(1));
   p5_y = sin(q(1))*(a3*cos(q(2)+q(3)) + a2*cos(q(2)) + a1) - d4*cos(q(1));
   p5_z = a3*sin(q(2)+q(3)) + a2*sin(q(2)) + d4*cos(q(2)+q(3)) + d1;
   p5 = [p5_x, p5_y, p5_z];
  
   set(h_arm, 'XData', [p0(1) p1(1) p2(1) p3(1) p4(1) p5(1)], ...
              'YData', [p0(2) p1(2) p2(2) p3(2) p4(2) p5(2)], ...
              'ZData', [p0(3) p1(3) p2(3) p3(3) p4(3) p5(3)]);
  
   while toc < t_total(k); end
end
